   CONFIG.FREQ_HZ {166666666} \

  connect_bd_net -net reset_core_peripheral_aresetn [get_bd_pins reset_core/peripheral_aresetn] [get_bd_pins core/aresetn] [get_bd_pins compute_to_ddr/s_axi_aresetn] [get_bd_pins host_to_control/m_axi_aresetn]

  save_bd_design
