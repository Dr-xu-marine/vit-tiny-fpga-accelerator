def enc(ratios):
 ms=[];rs=[]
 for x in np.asarray(ratios).reshape(-1):
  f=Fraction.from_float(float(x))
  if f<=0:raise ValueError('nonpositive ratio')
  for r in range(62,-1,-1):
   m=round(f*(1<<r))
   if 0<m<1<<31:ms.append(m);rs.append(r);break
  else:raise OverflowError('U31 scale')
 return np.asarray(ms,dtype=np.uint32),np.asarray(rs,dtype=np.uint8)
