proc stage {name} {
 set f [open [file join $::out STAGE.txt] w];puts $f $name;close $f
 puts "HOST75_SYNTH_STAGE $name";flush stdout
}
set failures {}
proc checked {name command} {
 if {[catch {uplevel 1 $command} msg options]} {
  lappend ::failures $name
  set f [open [file join $::out ${name}_error.txt] w];puts $f $msg;puts $f [dict get $options -errorinfo];close $f
  puts $::checks "$name\tFAIL\t[string map {\n { } \t { }} $msg]"
 } else {puts $::checks "$name\tPASS"}
 flush $::checks
}

checked DSP128 {
 set pe [get_cells -hier -quiet -filter {REF_NAME == DSP48E2 && NAME =~ */gemm/tile/core/*}]
 if {[llength $pe]!=128} {error "Expected128 MAC DSPs, actual[llength $pe]"}

 foreach d $pe {
  set row [list $d]
  foreach k {AREG BREG MREG PREG} {set v [get_property $k $d];lappend row $v;if {$v!=1} {error "DSP register mismatch: $d $k=$v"}}
  puts $f [join $row "\t"]
 }
 close $f
}
checked blackboxes {
