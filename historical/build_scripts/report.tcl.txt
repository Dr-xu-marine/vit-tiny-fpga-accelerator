proc checked {name command} {
 global failures summary
 stage $name
 if {[catch {uplevel 1 $command} msg options]} {
  lappend failures $name
  puts $summary "$name\tFAIL\t[string map {\n { } \t { }} $msg]"
  set f [open [file join $::out ${name}_error.txt] w];puts $f $msg;puts $f [dict get $options -errorinfo];close $f
 } else {puts $summary "$name\tPASS"}
 flush $summary
}
proc prop_or_unknown {object property} {
 if {[llength $object]!=1 || $property ni [list_property $object]} {return UNKNOWN}
 return [get_property $property $object]
}
proc physical {pin_or_cell} {
 set c [get_cells -quiet $pin_or_cell]
 if {[llength $c]!=1} {
  set p [get_pins -quiet $pin_or_cell]
  if {[llength $p]==1} {set c [get_cells -quiet -of_objects $p]}
 }
 if {[llength $c]!=1} {return {UNKNOWN UNKNOWN UNKNOWN UNKNOWN}}
 set loc [prop_or_unknown $c LOC];set region UNKNOWN;set slr UNKNOWN
 if {$loc ne "" && $loc ne "UNKNOWN"} {
  set site [get_sites -quiet $loc]
  if {[llength $site]==1} {
   if {![catch {get_clock_regions -quiet -of_objects $site} q] && [llength $q]} {set region $q}
   if {![catch {get_slrs -quiet -of_objects $site} q] && [llength $q]} {set slr $q}
  }
 }
 return [list $c $loc $region $slr]
}
proc write_path_rows {paths file {include_physical 1}} {
 set f [open $file w]
 puts $f "slack_ns\tstartpoint\tendpoint\tpath_group\tlogic_levels\tdata_delay_ns\tstart_cell\tstart_LOC\tstart_CR\tstart_SLR\tend_cell\tend_LOC\tend_CR\tend_SLR"
 foreach p $paths {
  set a [prop_or_unknown $p STARTPOINT_PIN];set b [prop_or_unknown $p ENDPOINT_PIN]
  set ap {NOT_QUERIED NOT_QUERIED NOT_QUERIED NOT_QUERIED};set bp $ap
  if {$include_physical} {set ap [physical $a];set bp [physical $b]}
  puts $f "[prop_or_unknown $p SLACK]\t$a\t$b\t[prop_or_unknown $p PATH_GROUP]\t[prop_or_unknown $p LOGIC_LEVELS]\t[prop_or_unknown $p DATAPATH_DELAY]\t[join $ap \t]\t[join $bp \t]"
 }
 close $f
}
proc negative_inventory {delay} {
 # One worst path per endpoint, both setup and hold separately; never silently truncate.
 set cap 250000
 set paths [get_timing_paths -quiet -delay_type $delay -slack_lesser_than 0 -max_paths $cap -nworst 1]
 # LOC/SLR are queried for the detailed worst/family paths, avoiding hundreds
 # of thousands of repeated physical queries in the complete endpoint list.
 write_path_rows $paths [file join $::out negative_${delay}_endpoints.tsv] 0
 set f [open [file join $::out negative_${delay}_count.tsv] w]
 puts $f "count\t[llength $paths]";puts $f "cap\t$cap";puts $f "truncated\t[expr {[llength $paths]>=$cap}]";close $f
 if {[llength $paths]>=$cap} {error "Negative endpoint inventory reached cap; incomplete, not accepted"}
}
proc physical_controls {} {
 set f [open [file join $::out local_valid_physical.tsv] w]
 puts $f "cell\tLOC\tclock_region\tSLR\tdriver\tnet\tload_pins"
 foreach c [get_cells -hier -quiet -filter {NAME =~ */local_read_v1_reg* && IS_SEQUENTIAL == 1}] {
  set p [get_pins -quiet -of_objects $c -filter {REF_PIN_NAME == Q}]
  set n [get_nets -quiet -segments -of_objects $p]
  puts $f "[join [physical $c] \t]\t$p\t$n\t[llength [get_pins -quiet -leaf -of_objects $n -filter {DIRECTION == IN}]]"
 }
 close $f
 set f [open [file join $::out DSP_BRAM_locations.tsv] w]
 puts $f "cell\tLOC\tclock_region\tSLR\tprimitive\tAREG\tBREG\tMREG\tPREG"
 foreach c [get_cells -hier -quiet -filter {REF_NAME == DSP48E2 || REF_NAME == RAMB36E2 || REF_NAME == RAMB18E2}] {
  puts $f "[join [physical $c] \t]\t[get_property REF_NAME $c]\t[prop_or_unknown $c AREG]\t[prop_or_unknown $c BREG]\t[prop_or_unknown $c MREG]\t[prop_or_unknown $c PREG]"
 }
 close $f
}
proc family_report {name args} {
 set f [open [file join $::out path_families.tsv] a]
 foreach delay {max min} {
  set paths [get_timing_paths -quiet -delay_type $delay -max_paths 10 -nworst 1 {*}$args]
  puts $f "$name\t$delay\t[llength $paths]\t[expr {[llength $paths] ? [get_property SLACK [lindex $paths 0]] : {NOT_PRESENT}}]"
  if {[llength $paths]} {
   report_timing -delay_type $delay -max_paths 10 -nworst 1 {*}$args -path_type full_clock_expanded -input_pins -file [file join $::out family_${name}_${delay}.rpt]
   write_path_rows $paths [file join $::out family_${name}_${delay}.tsv]
  }
 }
 close $f
}
proc all_families {} {
 set f [open [file join $::out path_families.tsv] w];puts $f "family\tdelay_type\tpaths\tworst_slack_ns";close $f
 set bram [get_cells -hier -quiet -filter {REF_NAME == RAMB36E2 || REF_NAME == RAMB18E2}]
 set dsp [get_cells -hier -quiet -filter {REF_NAME == DSP48E2}]
 set do_pins [get_pins -quiet -of_objects $bram -filter {DIRECTION == OUT && (REF_PIN_NAME =~ DO* || REF_PIN_NAME =~ DOP*)}]
 set di_pins [get_pins -quiet -of_objects $dsp -filter {DIRECTION == IN && (REF_PIN_NAME =~ A* || REF_PIN_NAME =~ B*)}]
 if {[llength $do_pins] && [llength $di_pins]} {family_report BRAM_to_DSP -from $do_pins -to $di_pins}
 set addr [get_pins -quiet -of_objects $bram -filter {DIRECTION == IN && (REF_PIN_NAME =~ ADDR* || REF_PIN_NAME =~ EN* || REF_PIN_NAME =~ WE*)}]
 if {[llength $addr]} {family_report BRAM_address_enable -to $addr}
 set ctrl [get_pins -quiet -of_objects $dsp -filter {DIRECTION == IN && (REF_PIN_NAME =~ CE* || REF_PIN_NAME =~ RST* || REF_PIN_NAME =~ OPMODE* || REF_PIN_NAME =~ ALUMODE*)}]
 if {[llength $ctrl]} {family_report DSP_control -to $ctrl}
 foreach {name filter} {
  MAC {IS_SEQUENTIAL == 1 && NAME =~ */gemm/tile/core/*}
  Bias_RQ {IS_SEQUENTIAL == 1 && NAME =~ */gemm/tile/rq/*}
  AUX_math {IS_SEQUENTIAL == 1 && NAME =~ */aux/arithmetic/*}
  scheduler {IS_SEQUENTIAL == 1 && NAME =~ */sequencer/*}
  writeback_move {IS_SEQUENTIAL == 1 && NAME =~ */tile/move/*}
 } {
  set cells [get_cells -hier -quiet -filter $filter]
  set pins [get_pins -quiet -of_objects $cells -filter {DIRECTION == IN && IS_CLOCK == 0}]
  if {[llength $pins]} {family_report $name -to $pins} else {
   set f [open [file join $::out path_families.tsv] a];puts $f "$name\tUNKNOWN\t0\tNOT_PRESENT";close $f
  }
 }
}
