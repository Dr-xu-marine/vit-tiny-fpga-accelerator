   if {[lsearch -exact [list_property $c] $p]>=0} {puts $f "$p\t[get_property $p $c]"}

 }

 }
 close $f
}

proc checked_report {name command} {
 global failures summary
 if {[catch {uplevel 1 $command} msg options]} {
  lappend failures $name
  puts $summary "$name\tFAIL\t[string map {\n { } \t { }} $msg]"
  set f [open [file join $::out ${name}_error.txt] w]
  puts $f $msg; puts $f [dict get $options -errorinfo]; close $f
 } else {puts $summary "$name\tPASS"}
 flush $summary
}
