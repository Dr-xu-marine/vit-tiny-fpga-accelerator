put('fpga-golden-check','SKILL.md','''
---
name: fpga-golden-check
description: Compare frozen byte-code outputs and locate the first mismatch with strict native-exit and dump gates. Use for FPGA Golden logs and captured host outputs, not for claiming full-network coverage from a local test.
---
# FPGA Golden checking
Read `references/evidence.md` before use. Work on copied evidence and bind exact source/data SHA.
1. Establish scope: local L1, affected-path L2, or complete-network L3. Missing scope is BLOCKED.
2. Check native launcher and kernel exits independently. Both exit0 plus any new crash dump is FAIL.
3. Run `scripts/check.py expected.hex actual.hex`; codes must be two hexadecimal digits, one per line.
4. Report count, mismatches, first index, unsigned and signed codes. Different lengths are FAIL.
5. Run a copied one-bit corruption test; expected detection is a checker PASS, not numerical PASS.
6. Require actual expected checkpoint/value/Host counts for L3. A cache hit or old source PASS grants no new numerical PASS.
Never write expected values into DUT storage or overwrite original traces. A matching label or top1 alone is insufficient.
''')
put('fpga-golden-check','scripts/check.py','''
import json,re,sys
from pathlib import Path
def codes(p):
 lines=Path(p).read_text().splitlines()
 if not lines or any(not re.fullmatch(r'[0-9a-fA-F]{2}',x.strip()) for x in lines):raise ValueError('INVALID_HEX_FORMAT')
 return [int(x,16) for x in lines]
def compare(a,b):
 bad=[i for i,(x,y) in enumerate(zip(a,b)) if x!=y]
 first=None
 if bad:
  i=bad[0];first={'index':i,'expected':a[i],'actual':b[i],'expected_signed':a[i] if a[i]<128 else a[i]-256,'actual_signed':b[i] if b[i]<128 else b[i]-256}
 return {'status':'PASS' if len(a)==len(b) and not bad else 'FAIL','expected_count':len(a),'actual_count':len(b),'mismatch_count':len(bad),'first_mismatch':first}
def native_gate(launcher,kernel,dumps):return 'PASS' if launcher==kernel==0 and not dumps else 'FAIL'
if __name__=='__main__':
 r=compare(codes(sys.argv[1]),codes(sys.argv[2]));print(json.dumps(r));sys.exit(0 if r['status']=='PASS' else 1)
''')

put('vivado-evidence-audit','SKILL.md','''
---
name: vivado-evidence-audit
description: Audit existing Vivado setup hold pulse width route and resource reports against a frozen candidate. Use for evidence review and stage-specific PASS decisions; does not run implementation or manufacture constraints.
---
# Vivado evidence audit
1. Bind part, source/DCP SHA, tool version and effective/generated clocks before reading results.
2. Parse raw timing summary with `scripts/check.py`; retain raw reports and all negative endpoints.
3. Require setup WNS>=0/TNS=0, hold WHS>=0/THS=0, pulse width margins>=0 and zero unrouted/error nets.
4. Inspect unconstrained paths, clock interaction, DRC and CDC separately. Warning presence is not automatic failure or automatic clearance; classify every warning family.
5. Audit actual DSP register properties and intended BRAM classes from mapped/routed cells, not RTL attributes alone.
6. Report scope precisely: PE, complete GEMM or full board. Bitstream generation never proves inference correctness.
Do not insert false paths/multicycle exceptions, relax a clock without changing its real generator, or reuse PASS across different netlists.
See `references/evidence.md`. This package reviews existing files only.
''')
put('vivado-evidence-audit','scripts/check.py','''
import json,re,sys
from pathlib import Path
def timing(text):
 pattern=r'^\\s*(-?\\d+\\.\\d+)\\s+(-?\\d+\\.\\d+)\\s+(\\d+)\\s+(\\d+)\\s+(-?\\d+\\.\\d+)\\s+(-?\\d+\\.\\d+)\\s+(\\d+)\\s+(\\d+)\\s+(-?\\d+\\.\\d+)\\s+(-?\\d+\\.\\d+)\\s+(\\d+)\\s+(\\d+)\\s*$'
 rows=re.findall(pattern,text,re.M)
 if len(rows)!=1:raise ValueError('TIMING_SUMMARY_ROW_NOT_UNIQUE')
 keys=['WNS','TNS','setup_fail','setup_total','WHS','THS','hold_fail','hold_total','WPWS','TPWS','pulse_fail','pulse_total']
 return dict(zip(keys,[int(x) if i%4 in (2,3) else float(x) for i,x in enumerate(rows[0])]))
def gate(t):
 return 'PASS_TIMING_ONLY' if all(t[x]>=0 for x in ['WNS','WHS','WPWS']) and all(t[x]==0 for x in ['TNS','THS','TPWS','setup_fail','hold_fail','pulse_fail']) else 'FAIL'
if __name__=='__main__':
 t=timing(Path(sys.argv[1]).read_text());print(json.dumps({'status':gate(t),'metrics':t,'DRC_CDC_source_identity':'REQUIRES_SEPARATE_AUDIT'}))
''')

---
name: fpga-board-diagnosis
description: Diagnose FPGA board inference in staged JTAG DDR ABI upload compute and result checks while preserving evidence and session continuity. Use after hardware download or a Golden mismatch; never silently reset or rerun a failed job.
---
# Staged board diagnosis
1. Bind bitstream/LTX/Host ABI/data SHA. Require exactly one intended hardware AXI object; stale handles are BLOCKED.
2. Read identity, calibration and busy/error/status flags. Preserve current results before any reset, reprogramming or upload.
3. Verify bounded data upload and readback before a controlled start. Separate transport PASS from numeric PASS.
4. Capture all 1000 outputs twice read-only; compare each code and read stability. Correct top1 does not close byte mismatches.
5. Record session continuity explicitly. If reset/reprogram/power cycle intervened, later DDR contents cannot establish the original job's cause.
6. Locate first divergence using exact-source local or scoped RTL tests. These exclude real DDR/CDC unless explicitly modeled.
Do not download, write memory, reset, call MIG tests or start inference without scope authorization. This distributed package contains only an offline session checker, not live JTAG commands.
''')
put('fpga-board-diagnosis','scripts/check.py','''
import json,sys
from pathlib import Path
def inspect(d):
 failed=d['initial_inference']['mismatch_values']!=0
 reset=bool(d.get('user_confirmed_intervening_action'))
 return {'numeric_status':'FAIL' if failed else 'PASS_CURRENT_CAPTURE_ONLY','root_cause':'UNKNOWN','post_reset_evidence_can_explain_original':False if reset else 'UNKNOWN','new_board_operations':'NOT_RUN'}
if __name__=='__main__':print(json.dumps(inspect(json.loads(Path(sys.argv[1]).read_text('utf8')))))
''')
