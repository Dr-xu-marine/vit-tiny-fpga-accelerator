FONT='C:/Windows/Fonts/msyh.ttc';S=3
class Canvas:
 def __init__(self,name,w=1000,h=440):
  self.name=name;self.w=w;self.h=h;self.svg=[];self.im=Image.new('RGB',(w*S,h*S),'white');self.d=ImageDraw.Draw(self.im)
 def box(self,x,y,w,h,label,fill='#edf3f8',stroke='#35546a',size=18):
  self.svg.append(f'<rect x="{x}" y="{y}" width="{w}" height="{h}" rx="7" fill="{fill}" stroke="{stroke}" stroke-width="1.5"/>')
  self.d.rounded_rectangle((x*S,y*S,(x+w)*S,(y+h)*S),radius=7*S,fill=fill,outline=stroke,width=4)
  self.text(x+w/2,y+h/2,label,size,anchor='middle')
 def text(self,x,y,label,size=18,color='#182a35',anchor='start'):
  lines=str(label).split('\n');font=ImageFont.truetype(FONT,size*S)
  start=y-(len(lines)-1)*size*.65
  for i,line in enumerate(lines):
   yy=start+i*size*1.3
   self.svg.append(f'<text x="{x}" y="{yy}" text-anchor="{anchor}" dominant-baseline="middle" font-family="Microsoft YaHei,sans-serif" font-size="{size}" fill="{color}">{html.escape(line)}</text>')
   self.d.text((x*S,yy*S),line,font=font,fill=color,anchor='mm' if anchor=='middle' else 'lm')
 def line(self,x1,y1,x2,y2,color='#4c6474',arrow=True,width=2):
  self.svg.append(f'<line x1="{x1}" y1="{y1}" x2="{x2}" y2="{y2}" stroke="{color}" stroke-width="{width}"/>')
  self.d.line((x1*S,y1*S,x2*S,y2*S),fill=color,width=width*S)
  if arrow:
   a=math.atan2(y2-y1,x2-x1);pts=[(x2,y2),(x2-10*math.cos(a)+5*math.sin(a),y2-10*math.sin(a)-5*math.cos(a)),(x2-10*math.cos(a)-5*math.sin(a),y2-10*math.sin(a)+5*math.cos(a))]
   self.svg.append('<polygon points="'+' '.join(f'{x},{y}' for x,y in pts)+f'" fill="{color}"/>')
   self.d.polygon([(int(x*S),int(y*S)) for x,y in pts],fill=color)
 def dot(self,x,y,r=2,color='#315f86'):
  self.svg.append(f'<circle cx="{x}" cy="{y}" r="{r}" fill="{color}" fill-opacity="0.65"/>')
  self.d.ellipse(((x-r)*S,(y-r)*S,(x+r)*S,(y+r)*S),fill=color)
 def save(self):
  (O/(self.name+'.svg')).write_text(f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {self.w} {self.h}" width="{self.w}" height="{self.h}"><rect width="100%" height="100%" fill="white"/>'+''.join(self.svg)+'</svg>',encoding='utf-8')
  self.im.save(O/(self.name+'.png'),dpi=(300,300))
