def setfont(st,name,sz,bold=False):
 st.font.name='Times New Roman';st.font.size=Pt(sz);st.font.bold=bold;st.font.color.rgb=RGBColor(0,0,0)
 pr=st.element.get_or_add_rPr();f=pr.find(qn('w:rFonts'))
 if f is None:f=OxmlElement('w:rFonts');pr.append(f)
 f.set(qn('w:eastAsia'),name)

for sn,sz in [('Title',24),('Subtitle',15),('Heading 1',16),('Heading 2',13),('Heading 3',11.5)]:
 setfont(d.styles[sn],'黑体',sz,True)

def clean(t):
 # Soft wrapping for long identifiers in narrow cells while keeping originals in evidence.
 return re.sub(r'[a-f0-9]{64}',lambda m:m[0][:32]+'\u200b'+m[0][32:],t)
def para(text,style=None):
 p=d.add_paragraph(clean(text),style)
 if not style:p.paragraph_format.first_line_indent=Cm(.74)
 return p
def mr(text):
 r=OxmlElement('m:r');t=OxmlElement('m:t');t.text=text;r.append(t);return r
def seq(parent,items):
 for item in items:parent.append(mr(item) if isinstance(item,str) else item)
 return parent
def sub(base,script):
 x=OxmlElement('m:sSub');seqx=OxmlElement('m:e');seq(seqx,[base]);x.append(seqx)
 z=OxmlElement('m:sub');seq(z,[script]);x.append(z);return x
def sup(base,power):
 x=OxmlElement('m:sSup');a=OxmlElement('m:e');seq(a,[base]);x.append(a)
 b=OxmlElement('m:sup');seq(b,[power]);x.append(b);return x
def frac(num,den):
 x=OxmlElement('m:f');a=OxmlElement('m:num');seq(a,num);x.append(a)
 b=OxmlElement('m:den');seq(b,den);x.append(b);return x
def root(items):
 x=OxmlElement('m:rad');pr=OxmlElement('m:radPr');hide=OxmlElement('m:degHide');hide.set(qn('m:val'),'1');pr.append(hide);x.append(pr)
 x.append(OxmlElement('m:deg'));b=OxmlElement('m:e');seq(b,items);x.append(b);return x
def equation(text):
 p=d.add_paragraph();p.alignment=WD_ALIGN_PARAGRAPH.CENTER
 p.paragraph_format.space_after=Pt(7)
 math=OxmlElement('m:oMath')
 if text.startswith('q_in'):
  items=[sub('q','in'),' = ',sub('u','RGB8'),' − 128，',sub('s','in'),' = ',frac(['2'],['255']),'，',sub('z','in'),' = 0']
 elif text.startswith('a_j'):
  items=[sub('a','j'),' = ',sub('Σ','k'),sub('x','k'),' ',sub('w','kj'),'；',sub('b','j'),' = ',sub('a','j'),' + ',sub('B','j')]
 elif text.startswith('y_j'):
  items=[sub('y','j'),' = sat8(RNE(',frac([sub('b','j'),' ',sub('M','j')],[sup('2',sub('R','j'))]),') + ',sub('Z','j'),')']
 elif text.startswith('W ='):
  items=['W = 192U − ',sup('T','2'),' + 2473901；root = ⌊',root(['W · ',sup('2','16')]),'⌋']
 elif text.startswith('inv ='):
  items=['inv = RNE(',frac([sup('2','53')],['root']),')；',sub('n','i'),' = RNE(',frac(['(192',sub('q','i'),' − T) inv'],[sup('2','25')]),')']
 else:items=[text]
 seq(math,items);p._p.append(math)
def table(b):
 cp=d.add_paragraph(b['caption'],'Caption');cp.paragraph_format.keep_with_next=True
 t=d.add_table(rows=1, cols=len(b['headers']));t.alignment=WD_TABLE_ALIGNMENT.CENTER;t.autofit=False
 widths=b.get('widths') or [16/len(b['headers'])]*len(b['headers'])
 for row in [b['headers']]+b['rows']:
  cells=t.rows[0].cells if row is b['headers'] else t.add_row().cells
  for i,txt in enumerate(row):
   c=cells[i];c.width=Cm(widths[i]);c.vertical_alignment=WD_CELL_VERTICAL_ALIGNMENT.CENTER
   pr=c._tc.get_or_add_tcPr();m=OxmlElement('w:tcMar')
   for edge,val in [('top','70'),('bottom','70'),('left','90'),('right','90')]:
    el=OxmlElement('w:'+edge);el.set(qn('w:w'),val);el.set(qn('w:type'),'dxa');m.append(el)
   pr.append(m)
   if row is b['headers']:
    sh=OxmlElement('w:shd');sh.set(qn('w:fill'),'EDEDED');pr.append(sh)
   p=c.paragraphs[0];p.paragraph_format.space_after=Pt(0);p.paragraph_format.line_spacing=1.13
   r=p.add_run(clean(txt));r.font.size=Pt(9.2);r.font.bold=row is b['headers']
  rr=t.rows[0] if row is b['headers'] else t.rows[-1]
  prop=rr._tr.get_or_add_trPr();el=OxmlElement('w:cantSplit');prop.append(el)
  if row is b['headers']:
   rep=OxmlElement('w:tblHeader');prop.append(rep)
 # Grid widths as well as cell widths.
 for gc,w in zip(t._tbl.tblGrid.gridCol_lst,widths):gc.set(qn('w:w'),str(int(Cm(w).twips)))
 borders=OxmlElement('w:tblBorders')
 for edge in ['top','left','bottom','right','insideH','insideV']:
  x=OxmlElement('w:'+edge);x.set(qn('w:val'),'single');x.set(qn('w:sz'),'4');x.set(qn('w:color'),'C8C8C8');borders.append(x)
 t._tbl.tblPr.append(borders)
 d.add_paragraph().paragraph_format.space_after=Pt(0)
